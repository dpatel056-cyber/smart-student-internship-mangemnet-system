    <!-- Admin Sidebar behaviour: mobile toggle, active-page highlighting, tooltips, accordion submenus.
         Sidebar is always fully expanded (no collapse/icon-only mode). -->
    <script>
        (function () {
            var sidebar = document.getElementById('dashboardSidebar');
            var toggleBtn = document.getElementById('sidebarToggleBtn');
            var backdrop = document.getElementById('sidebarBackdrop');
            if (!sidebar) return;

            /* ---- Mobile hamburger open / close ---- */
            function openMobileSidebar() {
                sidebar.classList.add('open');
                if (backdrop) backdrop.classList.add('show');
                document.body.style.overflow = 'hidden';
            }
            function closeMobileSidebar() {
                sidebar.classList.remove('open');
                if (backdrop) backdrop.classList.remove('show');
                document.body.style.overflow = '';
            }
            if (toggleBtn) {
                toggleBtn.addEventListener('click', function () {
                    sidebar.classList.contains('open') ? closeMobileSidebar() : openMobileSidebar();
                });
            }
            if (backdrop) backdrop.addEventListener('click', closeMobileSidebar);

            /* Close the mobile sidebar automatically after a nav link is tapped */
            sidebar.querySelectorAll('.asb-link[href], .asb-sublink[href]').forEach(function (a) {
                a.addEventListener('click', function () {
                    if (window.innerWidth <= 992) closeMobileSidebar();
                });
            });

            /* ---- Active page highlighting (link only — submenus stay CLOSED on load) ---- */
            var currentPage = (window.location.pathname.split('/').pop() || 'admin-dashboard.aspx').toLowerCase();
            if (!currentPage) currentPage = 'admin-dashboard.aspx';

            sidebar.querySelectorAll('.asb-link[data-page]').forEach(function (a) {
                if (a.getAttribute('data-page').toLowerCase() === currentPage) a.classList.add('active');
            });

            sidebar.querySelectorAll('.asb-sublink[data-page]').forEach(function (a) {
                if (a.getAttribute('data-page').toLowerCase() === currentPage) a.classList.add('active');
            });

            /* =========================================================
               ACCORDION (pure vanilla JS)
               - Every submenu starts CLOSED (no .open class in the markup).
               - Clicking a group button opens ONLY that submenu.
               - Opening one submenu closes any other submenu that was open.
               - Clicking the currently-open group's button again closes it.
               - Smooth animation via the CSS max-height/opacity transition
                 on .asb-submenu, driven by setting an inline max-height.
            ========================================================= */
            var groupButtons = Array.prototype.slice.call(sidebar.querySelectorAll('.asb-group-btn'));

            function closeGroup(btn) {
                var submenu = document.getElementById(btn.getAttribute('data-target'));
                if (!submenu) return;
                submenu.style.maxHeight = '0px';
                submenu.classList.remove('open');
                btn.setAttribute('aria-expanded', 'false');
                btn.classList.remove('active-parent');
            }

            function openGroup(btn) {
                var submenu = document.getElementById(btn.getAttribute('data-target'));
                if (!submenu) return;
                submenu.classList.add('open');
                submenu.style.maxHeight = submenu.scrollHeight + 'px';
                btn.setAttribute('aria-expanded', 'true');
                btn.classList.add('active-parent');
            }

            groupButtons.forEach(function (btn) {
                // Ensure a clean closed starting state regardless of any markup leftovers
                closeGroup(btn);

                btn.addEventListener('click', function () {
                    var submenu = document.getElementById(btn.getAttribute('data-target'));
                    var isOpen = submenu && submenu.classList.contains('open');

                    // Close every group first (max one open at a time)
                    groupButtons.forEach(closeGroup);

                    // If it wasn't already open, open it now (this also makes
                    // clicking an already-open group simply close it, since we
                    // just closed everything above and skip re-opening here).
                    if (!isOpen) openGroup(btn);
                });
            });

            /* Keep open submenu's height correct if the window is resized
               (e.g. rotating a tablet) while it's expanded. */
            window.addEventListener('resize', function () {
                var openBtn = groupButtons.filter(function (b) { return b.classList.contains('active-parent'); })[0];
                if (!openBtn) return;
                var submenu = document.getElementById(openBtn.getAttribute('data-target'));
                if (submenu) submenu.style.maxHeight = submenu.scrollHeight + 'px';
            });
        })();
    </script>
